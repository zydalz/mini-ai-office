// ==========================================
// GRAVITATIONAL ORBIT CALCULATOR v1.0
// Created by Spark (Dev) & Approved by Echo (QA)
// ==========================================

const G = 6.67430e-11; // Gravitational constant (m^3 kg^-1 s^-2)

export function calculateGravityForce(mass1, mass2, radius) {
  // Edge-case safeguard (audited by Echo)
  if (radius <= 0) {
    throw new Error("Radius must be greater than zero.");
  }
  
  const force = (G * mass1 * mass2) / Math.pow(radius, 2);
  return {
    forceNewtons: force,
    formula: "F = G * (m1 * m2) / r^2",
    status: "APPROVED"
  };
}
